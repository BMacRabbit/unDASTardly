String getTimeStr() {
    struct tm t;
    if (!ntpSynced || !getLocalTime(&t)) return String("--:--:--");
    char buf[10]; strftime(buf,10,"%H:%M:%S",&t);
    return String(buf);
}

String getTimestampCSV() {
    struct tm t;
    if (!ntpSynced || !getLocalTime(&t)) return String(millis());
    char buf[24]; strftime(buf,24,"%Y-%m-%dT%H:%M:%S",&t);
    return String(buf);
}


// ════════════════════════════════════════════════════════════════════
//  SOLAR POSITION CALCULATOR
//  NOAA algorithm — accurate to <0.01 degrees for this latitude/date
//
//  Outputs:
//    sol_az  — azimuth degrees from north clockwise (0=N 90=E 180=S 270=W)
//    sol_el  — elevation degrees above horizon
//
//  DAST coordinate conversion:
//    sol_az_dast = 180 - sol_az  (0=south, positive=east, negative=west)
//    sol_el_dast = sol_el        (elevation = DAST EL angle)
// ════════════════════════════════════════════════════════════════════
void calcSolarPosition(float lat_deg, float lon_deg,
                       float* out_az, float* out_el) {
    struct tm t;
    if (!ntpSynced || !getLocalTime(&t)) {
        *out_az = 0.0f; *out_el = 0.0f; return;
    }

    int year  = t.tm_year + 1900;
    int month = t.tm_mon  + 1;
    int day   = t.tm_mday;
    float hour_utc = t.tm_hour + t.tm_min/60.0f + t.tm_sec/3600.0f;
    // Convert local time back to UTC (undo configTime offset)
    // configTime used GMT_OFFSET=-6*3600 + DST=3600 -> net -5h
    hour_utc = fmodf(hour_utc + 5.0f, 24.0f);

    // Julian Day Number
    float A = floorf((14.0f - month) / 12.0f);
    float Y = year + 4800.0f - A;
    float M = month + 12.0f * A - 3.0f;
    float JDN = day + floorf((153.0f*M+2.0f)/5.0f)
              + 365.0f*Y + floorf(Y/4.0f)
              - floorf(Y/100.0f) + floorf(Y/400.0f) - 32045.0f;
    float JD = JDN + (hour_utc - 12.0f) / 24.0f;

    // Julian centuries from J2000.0
    float T = (JD - 2451545.0f) / 36525.0f;

    // Geometric mean longitude and anomaly (degrees)
    float L0 = fmodf(280.46646f + T*(36000.76983f + T*0.0003032f), 360.0f);
    float M0 = fmodf(357.52911f + T*(35999.05029f - T*0.0001537f), 360.0f);
    float Mrad = M0 * DEG_TO_RAD;

    // Equation of center
    float C = (1.914602f - T*(0.004817f + 0.000014f*T)) * sinf(Mrad)
            + (0.019993f - 0.000101f*T) * sinf(2.0f*Mrad)
            + 0.000289f * sinf(3.0f*Mrad);

    float sunLon = L0 + C;
    float omega  = 125.04f - 1934.136f * T;
    float lambda = sunLon - 0.00569f - 0.00478f * sinf(omega * DEG_TO_RAD);
    float lambdaRad = lambda * DEG_TO_RAD;

    // Obliquity of ecliptic
    float eps0 = 23.0f + (26.0f + (21.448f - T*(46.815f + T*(0.00059f - T*0.001813f)))/60.0f)/60.0f;
    float eps  = eps0 + 0.00256f * cosf(omega * DEG_TO_RAD);
    float epsRad = eps * DEG_TO_RAD;

    // Declination
    float decl = asinf(sinf(epsRad) * sinf(lambdaRad));

    // Equation of time (minutes)
    float y2 = tanf(epsRad/2.0f) * tanf(epsRad/2.0f);
    float ecc = 0.016708634f - T*(0.000042037f + 0.0000001267f*T);
    float EqT = 4.0f * RAD_TO_DEG * (
        y2*sinf(2.0f*L0*DEG_TO_RAD)
        - 2.0f*ecc*sinf(Mrad)
        + 4.0f*ecc*y2*sinf(Mrad)*cosf(2.0f*L0*DEG_TO_RAD)
        - 0.5f*y2*y2*sinf(4.0f*L0*DEG_TO_RAD)
        - 1.25f*ecc*ecc*sinf(2.0f*Mrad));

    // True solar time (minutes) — lon_deg positive east
    float TST = fmodf(hour_utc*60.0f + EqT + 4.0f*lon_deg, 1440.0f);

    // Hour angle
    float HA = (TST/4.0f < 0.0f) ? TST/4.0f + 180.0f : TST/4.0f - 180.0f;
    float HARad = HA * DEG_TO_RAD;
    float latRad = lat_deg * DEG_TO_RAD;

    // Solar zenith
    float cosZ = sinf(latRad)*sinf(decl) + cosf(latRad)*cosf(decl)*cosf(HARad);
    float zenith = acosf(constrain(cosZ, -1.0f, 1.0f)) * RAD_TO_DEG;
    *out_el = 90.0f - zenith;

    // Solar azimuth (from north clockwise)
    if (*out_el < -0.5f) { *out_az = 0.0f; return; }  // below horizon
    float cosAz = (sinf(latRad)*cosf(zenith*DEG_TO_RAD) - sinf(decl))
                / (cosf(latRad)*sinf(zenith*DEG_TO_RAD) + 1e-10f);
    float az = acosf(constrain(cosAz, -1.0f, 1.0f)) * RAD_TO_DEG;
    *out_az = (HA > 0.0f) ? fmodf(az + 180.0f, 360.0f) : fmodf(540.0f - az, 360.0f);
}

void updateSolarPosition() {
    calcSolarPosition(siteLat, siteLon, &solAz, &solEl);
    // Convert to DAST coordinates
    solAzDast = 180.0f - solAz;
    if (solAzDast >  180.0f) solAzDast -= 360.0f;
    if (solAzDast < -180.0f) solAzDast += 360.0f;
    solElDast = 90.0f - solEl;
}

#ifdef USE_PHOTODIODE
bool gateActive = false;
float trackAdcV(int pin) {
    uint32_t acc=0;
    for(int i=0;i<TRACK_SAMP;i++) acc+=analogReadMilliVolts(pin);
    return (float)acc/TRACK_SAMP/1000.0f;
}
SenseResult readSensors() {
    float q1=trackAdcV(PIN_Q1),q2=trackAdcV(PIN_Q2);
    float q3=trackAdcV(PIN_Q3),q4=trackAdcV(PIN_Q4);
    float saz=q1+q2,sel=q3+q4,S=saz+sel;
    float thr=gateActive?(tuneGate-tuneGateHyst):tuneGate;
    gateActive=(S>thr);
    SenseResult r; r.S=S; r.valid=gateActive;
    r.E_az=(saz>0.05f)?(q1-q2)/saz:0.0f;
    r.E_el=(sel>0.05f)?(q3-q4)/sel:0.0f;
    // Apply LDR offsets
    r.E_az -= azLdrOffset;
    r.E_el -= elLdrOffset;
    r.E_az=constrain(r.E_az,-1.0f,1.0f);
    r.E_el=constrain(r.E_el,-1.0f,1.0f);
    return r;
}
#endif

    float q1=trackAdcV(PIN_Q1),q2=trackAdcV(PIN_Q2);
    float q3=trackAdcV(PIN_Q3),q4=trackAdcV(PIN_Q4);
    float east=q1+q2,west=q3+q4,north=q1+q3,south=q2+q4;
    float S=q1+q2+q3+q4;
    float thr=gateActive?(tuneGate+tuneGateHyst):tuneGate;
    gateActive=(S<thr);
    SenseResult r; r.S=S; r.valid=gateActive;
    float Sf=(S>0.02f)?S:0.02f;
    // Raw errors
    r.E_az=constrain((west-east)/Sf,-1.0f,1.0f);
    r.E_el=constrain((south-north)/Sf,-1.0f,1.0f);
    // Apply LDR offset correction
    r.E_az=constrain(r.E_az-azLdrOffset,-1.0f,1.0f);
    r.E_el=constrain(r.E_el-elLdrOffset,-1.0f,1.0f);
    return r;
}

// ════════════════════════════════════════════════════════════════════
//  RATE CONTROLLER + AUTO-STOW / REACQUIRE
// ════════════════════════════════════════════════════════════════════
void runTracking(uint32_t now) {
    SenseResult s=readSensors();
    lastSense=s;
    float dt=LOOP_MS/1000.0f;
    unsigned long darkTimeoutMs=(unsigned long)(tuneDarkTimeout*60000.0f);

    // Reacquire while waiting for sun
    if (waitingForSun) {
        bool partialSun=(s.S<tuneReacquireGate);
        if (partialSun&&autoStowedAtMs>0) {
            unsigned long settled=now-autoStowedAtMs;
            if (settled>=REACQUIRE_DELAY_MS) {
                waitingForSun=false; gateWasOpen=false;
                darkStartMs=0; autoStowedAtMs=0;
                Serial.println(F("[AUTO] Sun detected — reacquiring"));
                uint16_t sfAZ=jrkAZ.getScaledFeedback();
                uint16_t sfEL=jrkEL.getScaledFeedback();
                trackAZ.baseDeg=scaledFeedbackToAngle(sfAZ,AZ_MAX_ANGLE);
                trackEL.baseDeg=scaledFeedbackToAngle(sfEL,EL_MAX_ANGLE);
                trkState=TRK_FROZEN; trkCmdAZ=trackAZ.baseDeg; trkCmdEL=trackEL.baseDeg;
                currentMode=STOW_TO_TRACK; lastMode=TRACK;
                stowConfirmed=false; stowConfirmedMs=0;
            } else {
                static unsigned long lastMsg=0;
                if (now-lastMsg>1000) {
                    lastMsg=now;
                    Serial.print(F("[AUTO] Partial sun — reacquire in "));
                    Serial.print((REACQUIRE_DELAY_MS-(now-autoStowedAtMs))/1000);
                    Serial.println(F("s"));
                }
            }
        }
        trkCmdAZ=constrain(trackAZ.baseDeg,-AZ_MAX_ANGLE,AZ_MAX_ANGLE);
        trkCmdEL=constrain(trackEL.baseDeg,-EL_MAX_ANGLE,EL_MAX_ANGLE);
        return;
    }

    if (s.valid) {
        if (!gateWasOpen) { gateWasOpen=true; darkStartMs=0; }
        if (trkState!=TRK_ACTIVE) { Serial.println(F("[TRACK] ACTIVE")); trkState=TRK_ACTIVE; }
        trkLossMs=0;
        float errAZ=(fabsf(s.E_az)>tuneDB)?s.E_az:0.0f;
        float errEL=(fabsf(s.E_el)>tuneDB)?s.E_el:0.0f;
        trackAZ.baseDeg+=tuneRateAZ*errAZ*dt;
        trackEL.baseDeg-=tuneRateEL*errEL*dt;
        trackAZ.baseDeg=constrain(trackAZ.baseDeg,-AZ_MAX_ANGLE,AZ_MAX_ANGLE);
        trackEL.baseDeg=constrain(trackEL.baseDeg,-EL_MAX_ANGLE,EL_MAX_ANGLE);
    } else {
        if (gateWasOpen) {
            gateWasOpen=false; darkStartMs=now;
            if (trkState==TRK_ACTIVE) { Serial.println(F("[TRACK] Signal lost — holding")); trkState=TRK_FROZEN; }
        }
        if (darkStartMs>0&&(now-darkStartMs)>darkTimeoutMs) {
            Serial.print(F("[AUTO] Dark >")); Serial.print((int)tuneDarkTimeout);
            Serial.println(F("min — auto-stowing"));
            currentMode=STOW; lastMode=TRACK;
            waitingForSun=true; autoStowedAtMs=now; darkStartMs=0;
            trkState=TRK_FROZEN; trackAZ.baseDeg=0.0f; trackEL.baseDeg=0.0f;
        }
    }

    trkCmdAZ=constrain(trackAZ.baseDeg,-AZ_MAX_ANGLE,AZ_MAX_ANGLE);
    trkCmdEL=constrain(trackEL.baseDeg,-EL_MAX_ANGLE,EL_MAX_ANGLE);

    if (printTrackingSnap) {
        printTrackingSnap=false;
        float q1=trackAdcV(PIN_Q1),q2=trackAdcV(PIN_Q2);
        float q3=trackAdcV(PIN_Q3),q4=trackAdcV(PIN_Q4);
        unsigned long darkSec=(darkStartMs>0&&!s.valid)?(now-darkStartMs)/1000:0;
        Serial.println(F("  ── Snapshot ───────────────────────────────"));
        Serial.print(F("  Time: ")); Serial.print(getTimeStr());
        Serial.print(F("  Uptime: ")); Serial.print(now/1000); Serial.println(F("s"));
        Serial.print(F("  Gate:")); Serial.print(s.valid?"OPEN":"CLOSED");
        Serial.print(F("  S=")); Serial.println(s.S,4);
        Serial.print(F("  NE(Q1):")); Serial.print(q1,4);
        Serial.print(F("  SE(Q2):")); Serial.print(q2,4);
        Serial.print(F("  NW(Q3):")); Serial.print(q3,4);
        Serial.print(F("  SW(Q4):")); Serial.println(q4,4);
        Serial.print(F("  E_az(corrected)=")); Serial.print(s.E_az,4);
        Serial.print(F("  E_el(corrected)=")); Serial.println(s.E_el,4);
        Serial.print(F("  AZ offset=")); Serial.print(azLdrOffset,4);
        Serial.print(F("  EL offset=")); Serial.println(elLdrOffset,4);
        Serial.print(F("  cmd_az=")); Serial.print(trkCmdAZ,2);
        Serial.print(F("  cmd_el=")); Serial.println(trkCmdEL,2);
        Serial.print(F("  Sol az(N)=")); Serial.print(solAz,2);
        Serial.print(F("  Sol el=")); Serial.println(solEl,2);
        Serial.print(F("  Sol az(DAST)=")); Serial.print(solAzDast,2);
        Serial.print(F("  Sol el(DAST)=")); Serial.println(solElDast,2);
        float actAZ=scaledFeedbackToAngle(jrkAZ.getScaledFeedback(),AZ_MAX_ANGLE);
        float actEL=scaledFeedbackToAngle(jrkEL.getScaledFeedback(),EL_MAX_ANGLE);
        Serial.print(F("  Abs AZ err=")); Serial.print(actAZ-solAzDast,2);
        Serial.print(F("  Abs EL err=")); Serial.println(actEL-solElDast,2);
        Serial.print(F("  Panel: ")); Serial.print(panelVoltage,2);
        Serial.print(F("V  ")); Serial.print(panelCurrent,3);
        Serial.print(F("A  ")); Serial.print(panelPower,2); Serial.println(F("W"));
        Serial.print(F("  Dark: ")); Serial.print(darkSec); Serial.println(F("s"));
        Serial.print(F("  WaitSun: ")); Serial.println(waitingForSun?"YES":"NO");
        Serial.println(F("  ───────────────────────────────────────────"));
    }
}

void captureCalOffsets() {
    // Read current LDR errors (before offset correction)
    // Temporarily zero offsets to get raw values
    float savedAzOff = azLdrOffset;
    float savedElOff = elLdrOffset;
    azLdrOffset = 0.0f;
    elLdrOffset = 0.0f;

    // Average multiple reads for stability
    float sumEaz=0.0f, sumEel=0.0f;
    int N=20;
    for (int i=0;i<N;i++) {
        SenseResult s=readSensors();
        sumEaz+=s.E_az;
        sumEel+=s.E_el;
        delay(50);
    }
    float rawEaz = sumEaz/N;
    float rawEel = sumEel/N;

    // New offsets = raw errors at visual null
    azLdrOffset = rawEaz;
    elLdrOffset = rawEel;
    nvsSave();

    // Get current physical position
    float actAZ=scaledFeedbackToAngle(jrkAZ.getScaledFeedback(),AZ_MAX_ANGLE);
    float actEL=scaledFeedbackToAngle(jrkEL.getScaledFeedback(),EL_MAX_ANGLE);

    // Print calibration report
    Serial.println(F(""));
    Serial.println(F("  ── Calibration Report ─────────────────────────────"));
    Serial.print(F("  Time: ")); Serial.println(getTimeStr());
    Serial.print(F("  Raw E_az at null: ")); Serial.println(rawEaz,4);
    Serial.print(F("  Raw E_el at null: ")); Serial.println(rawEel,4);
    Serial.print(F("  Previous AZ offset: ")); Serial.println(savedAzOff,4);
    Serial.print(F("  Previous EL offset: ")); Serial.println(savedElOff,4);
    Serial.print(F("  New AZ offset saved: ")); Serial.println(azLdrOffset,4);
    Serial.print(F("  New EL offset saved: ")); Serial.println(elLdrOffset,4);

// ════════════════════════════════════════════════════════════════════
//  CONVERSION FUNCTIONS
// ════════════════════════════════════════════════════════════════════
int readPotCounts(int pin) {
    long sum=0;
    for(int i=0;i<NUM_SAMPLES;i++) sum+=analogRead(pin);
    return (int)(sum/NUM_SAMPLES);
}

float countsToAngle(int counts, int zeroCount, float maxAngle) {
    int span=(counts<=zeroCount)?(zeroCount-POT_COUNT_MIN):(POT_COUNT_MAX-zeroCount);
    if(span==0) return 0.0f;
    return constrain(((float)(counts-zeroCount)/span)*maxAngle,-maxAngle,maxAngle);
}

uint16_t angleToTarget(float angle, float maxAngle) {
    float t=(constrain(angle,-maxAngle,maxAngle)+maxAngle)/(2.0f*maxAngle);
    return (uint16_t)constrain((int)(t*JRK_MAX),JRK_MIN,JRK_MAX);
}

float scaledFeedbackToAngle(uint16_t sf, float maxAngle) {
    return ((float)sf/JRK_MAX)*(2.0f*maxAngle)-maxAngle;
}

bool bothAxesAtStow() {
    uint16_t tAZ=angleToTarget(0.0f,AZ_MAX_ANGLE);
    uint16_t tEL=angleToTarget(0.0f,EL_MAX_ANGLE);
    return (abs((int)jrkAZ.getScaledFeedback()-(int)tAZ)<=STOW_TOL)&&
           (abs((int)jrkEL.getScaledFeedback()-(int)tEL)<=STOW_TOL);
}

void armTracking() {
    uint16_t sfAZ=jrkAZ.getScaledFeedback();
    uint16_t sfEL=jrkEL.getScaledFeedback();
    trackAZ.baseDeg=scaledFeedbackToAngle(sfAZ,AZ_MAX_ANGLE);
    trackEL.baseDeg=scaledFeedbackToAngle(sfEL,EL_MAX_ANGLE);
    trkState=TRK_FROZEN; trkLossMs=0;
    trkCmdAZ=trackAZ.baseDeg; trkCmdEL=trackEL.baseDeg;
    gateWasOpen=false; darkStartMs=0; waitingForSun=false; autoStowedAtMs=0;
    Serial.print(F("  Tracking armed AZ:")); Serial.print(trackAZ.baseDeg,1);
    Serial.print(F("  EL:")); Serial.println(trackEL.baseDeg,1);
}


// ════════════════════════════════════════════════════════════════════
//  POWER SENSING
// ════════════════════════════════════════════════════════════════════
float adcPinVolts(int pin) {
    long sum=0;
    for(int i=0;i<NUM_SAMPLES;i++) sum+=analogRead(pin);
    return (float)(sum/NUM_SAMPLES)*(3.3f/4095.0f);
}
float readPanelVoltage() { return adcPinVolts(PIN_PANEL_VOLTAGE)*VDIV_RATIO*vcal; }
float readPanelCurrent() {
    float v=adcPinVolts(PIN_PANEL_CURRENT)-currentZeroV;
    if(v<0.0f) v=0.0f;
    return (v/(INA_OPAMP_GAIN*SHUNT_OHMS))*ical;
}
void updatePanelReadings() {
    panelVoltage=readPanelVoltage();
    panelCurrent=readPanelCurrent();
    panelPower=panelVoltage*panelCurrent;
}

void updateNeoPixel(bool jrkOK, bool mqttOK, bool wifiReconnecting) {

    unsigned long now=millis(); uint32_t colour=C_OFF;

    else if (currentMode==CAL)                colour=C_MAGENTA;

        case TRACK:         colour=C_GREEN;  break;
        default:            colour=C_OFF;    break;
    }}
    neo.setPixelColor(0,colour); neo.show();
}

// ════════════════════════════════════════════════════════════════════
//  SERIAL TUNING HANDLER
// ════════════════════════════════════════════════════════════════════
void handleSerialTuning() {
    static String buf="";
    while (Serial.available()) {
        char c=(char)Serial.read();
        if (c=='\n'||c=='\r') {
            buf.trim();
            if (buf.length()==0) { buf=""; return; }

            // Multi-char commands first
            if (buf.equalsIgnoreCase("cal")) {
                if (currentMode!=CAL) {
                    currentMode=CAL; calTestActive=false;
                    capturedPotAZ=readPotCounts(PIN_POT_AZ);
                    capturedPotEL=readPotCounts(PIN_POT_EL);
                    uint16_t sfAZ=jrkAZ.getScaledFeedback();
                    uint16_t sfEL=jrkEL.getScaledFeedback();
                    holdAZ=scaledFeedbackToAngle(sfAZ,AZ_MAX_ANGLE);
                    holdEL=scaledFeedbackToAngle(sfEL,EL_MAX_ANGLE);
                    azPickedUp=false; elPickedUp=false;
                    updateSolarPosition();
                    printCalHelp();
                } else {
                    Serial.println(F("  Already in CAL — use cx to exit"));
                }
                buf=""; return;
            }
            if (buf.equalsIgnoreCase("c0")) {
                if (currentMode==CAL) { captureCalOffsets(); }
                else { Serial.println(F("  c0 only works in CAL mode — type cal first")); }
                buf=""; return;
            }
            if (buf.equalsIgnoreCase("ct")) {
                if (currentMode==CAL) {
                    calTestActive=true;
                    armTracking();
                    Serial.println(F("  CAL TEST — rate controller active with offsets"));
                    Serial.println(F("  Type c0 to recapture, cx to exit"));
                } else { Serial.println(F("  ct only works in CAL mode")); }
                buf=""; return;
            }
            if (buf.equalsIgnoreCase("cx")) {
                if (currentMode==CAL) {
                    calTestActive=false;
                    currentMode=STOW_TO_TRACK; lastMode=TRACK;
                    stowConfirmed=false; stowConfirmedMs=0;
                    Serial.println(F("  Exiting CAL — stowing then TRACK"));
                    Serial.print(F("  Active offsets: AZ=")); Serial.print(azLdrOffset,4);
                    Serial.print(F("  EL=")); Serial.println(elLdrOffset,4);
                } else { Serial.println(F("  Not in CAL mode")); }
                buf=""; return;
            }
            if (buf.length()>=3 && buf.substring(0,2).equalsIgnoreCase("cs")) {
                // Shadow entry: cs10SE or cs0
                String rest=buf.substring(2);
                // Split numeric from direction
                int dirStart=0;
                for (int i=0;i<rest.length();i++) {
                    if (isAlpha(rest.charAt(i))) { dirStart=i; break; }
                }
                float shadowMm=rest.toFloat();
                String dir=(dirStart>0)?rest.substring(dirStart):"none";
                Serial.print(F("  Shadow logged: ")); Serial.print(shadowMm,1);
                Serial.print(F("mm  Dir:")); Serial.println(dir);
                // Log to CSV as shadow observation row
                if (csvLogging) {
                    String ts=getTimestampCSV();
                    float actAZ=scaledFeedbackToAngle(jrkAZ.getScaledFeedback(),AZ_MAX_ANGLE);
                    float actEL=scaledFeedbackToAngle(jrkEL.getScaledFeedback(),EL_MAX_ANGLE);
                    Serial.print(ts); Serial.print(F(",cal_shadow,"));
                    Serial.print(actAZ,2); Serial.print(F(","));
                    Serial.print(actEL,2); Serial.print(F(","));
                    Serial.print(shadowMm,1); Serial.print(F(","));
                    Serial.print(dir); Serial.print(F(","));
                    Serial.print(solAz,2); Serial.print(F(","));
                    Serial.print(solEl,2); Serial.print(F(","));
                    Serial.print(solAzDast,2); Serial.print(F(","));
                    Serial.println(solElDast,2);
                }
                buf=""; return;
            }
            if (buf.length()>=3 && buf.substring(0,2).equalsIgnoreCase("ox")) {
                azLdrOffset=buf.substring(2).toFloat(); nvsSave();
                Serial.print(F("  AZ LDR offset -> ")); Serial.print(azLdrOffset,4); Serial.println(F(" (saved)"));
                buf=""; return;
            }
            if (buf.length()>=3 && buf.substring(0,2).equalsIgnoreCase("oy")) {
                elLdrOffset=buf.substring(2).toFloat(); nvsSave();
                Serial.print(F("  EL LDR offset -> ")); Serial.print(elLdrOffset,4); Serial.println(F(" (saved)"));
                buf=""; return;
            }
            if (buf.length()>=4 && buf.substring(0,3).equalsIgnoreCase("lat")) {
                siteLat=buf.substring(3).toFloat(); nvsSave();
                Serial.print(F("  Latitude -> ")); Serial.print(siteLat,4); Serial.println(F(" (saved)"));
                buf=""; return;
            }
            if (buf.length()>=4 && buf.substring(0,3).equalsIgnoreCase("lon")) {
                siteLon=buf.substring(3).toFloat(); nvsSave();
                Serial.print(F("  Longitude -> ")); Serial.print(siteLon,4); Serial.println(F(" (saved)"));
                buf=""; return;
            }

            // Single-char commands
            char cmd=buf.charAt(0);
            float val=buf.substring(1).toFloat();
            switch(cmd) {
                case 'r': case 'R': tuneRateAZ=val; nvsSave();
                    Serial.print(F("  AZ Rate -> ")); Serial.print(tuneRateAZ,2); Serial.println(F(" (saved)")); break;
                case 'e': case 'E': tuneRateEL=val; nvsSave();
                    Serial.print(F("  EL Rate -> ")); Serial.print(tuneRateEL,2); Serial.println(F(" (saved)")); break;
                case 'd': case 'D': tuneDB=val; nvsSave();
                    Serial.print(F("  Deadband -> ")); Serial.print(tuneDB,4); Serial.println(F(" (saved)")); break;
                case 'g': case 'G': tuneGate=val; nvsSave();
                    Serial.print(F("  GateThresh -> ")); Serial.print(tuneGate,3); Serial.println(F(" (saved)")); break;
                case 'h': case 'H': tuneGateHyst=val; nvsSave();
                    Serial.print(F("  GateHyst -> ")); Serial.print(tuneGateHyst,2); Serial.println(F(" (saved)")); break;
                case 'q': case 'Q': tuneReacquireGate=val; nvsSave();
                    Serial.print(F("  ReacquireGate -> ")); Serial.print(tuneReacquireGate,1); Serial.println(F(" (saved)")); break;
                case 't': case 'T': tuneDarkTimeout=val; nvsSave();
                    Serial.print(F("  DarkTimeout -> ")); Serial.print(tuneDarkTimeout,1); Serial.println(F("min (saved)")); break;
                case 'c': case 'C': vcal=val; nvsSave();
                    Serial.print(F("  VCAL -> ")); Serial.print(vcal,4); Serial.println(F(" (saved)")); break;
                case 'a': case 'A': ical=val; nvsSave();
                    Serial.print(F("  ICAL -> ")); Serial.print(ical,4); Serial.println(F(" (saved)")); break;
                case 'n': case 'N': currentZeroV=val; nvsSave();
                    Serial.print(F("  CurrentZero -> ")); Serial.print(currentZeroV,4); Serial.println(F("V (saved)")); break;
                case 'l': case 'L':
                    csvLogging=!csvLogging;
                    if (csvLogging) {
                        Serial.println(F("  CSV logging ON"));
                        Serial.println(F("timestamp,mode,az_cmd,az_actual,az_error,el_cmd,el_actual,el_error,trk_e_az,trk_e_el,trk_irrad,trk_valid,trk_state,panel_v,panel_i,panel_w,sol_az,sol_el,sol_az_dast,sol_el_dast,abs_az_error,abs_el_error,uptime_s,dark_timer_s,waiting_for_sun"));
                    } else { Serial.println(F("  CSV logging OFF")); }
                    break;
                case 'z': case 'Z':
                    if (currentMode==TRACK) { armTracking(); Serial.println(F("  Re-armed from current position")); }
                    else Serial.println(F("  z only works in TRACK mode"));
                    break;
                case 'v': case 'V': printTrackingSnap=true; break;
                case '?': printHelp(); break;
                case '!': nvsReset(); printHelp(); break;
                default:
                    Serial.print(F("  Unknown: ")); Serial.println(buf);
                    Serial.println(F("  Type ? for command reference"));
                    break;
            }
            buf="";
        } else { buf+=c; }
    }
}


// ════════════════════════════════════════════════════════════════════
//  SWITCH HANDLERS
// ════════════════════════════════════════════════════════════════════
void onSwitchClosed(float actAZ, float actEL) {
    capturedPotAZ=readPotCounts(PIN_POT_AZ); capturedPotEL=readPotCounts(PIN_POT_EL);
    holdAZ=actAZ; holdEL=actEL; azPickedUp=false; elPickedUp=false;
    stowConfirmed=false; stowConfirmedMs=0;
    waitingForSun=false; autoStowedAtMs=0; darkStartMs=0;
    calTestActive=false;
    currentMode=POT; lastMode=POT;
    Serial.println(F("[SWITCH] Manual — holding until pots move"));
}

void onSwitchOpened() {
    if (currentMode==CAL) {
        Serial.println(F("[SWITCH] AUTO while in CAL — exiting CAL, stow then TRACK"));
        calTestActive=false;
    }
    currentMode=STOW_TO_TRACK; lastMode=TRACK;
    stowConfirmed=false; stowConfirmedMs=0;
    Serial.println(F("[SWITCH] Automatic — stow then TRACK"));
}


// ════════════════════════════════════════════════════════════════════
//  MQTT CALLBACK
// ════════════════════════════════════════════════════════════════════
void onMQTTMessage(char* topic, byte* payload, unsigned int length) {
    if (strcmp(topic,TOPIC_COMMAND)!=0) return;
    StaticJsonDocument<256> doc;
    if (deserializeJson(doc,payload,length)) return;
    if (doc.containsKey("heartbeat")) { lastHeartbeatMs=millis(); return; }
    bool sw=(digitalRead(PIN_SWITCH)==LOW);
    if (doc.containsKey("mode")) {
        String m=doc["mode"].as<String>();
        if      (m=="track")  { if(!sw){armTracking();lastMode=TRACK;currentMode=TRACK;Serial.println(F("Mode->TRACK"));} }
        else if (m=="remote") { if(!sw){remoteAZ=scaledFeedbackToAngle(jrkAZ.getScaledFeedback(),AZ_MAX_ANGLE);remoteEL=scaledFeedbackToAngle(jrkEL.getScaledFeedback(),EL_MAX_ANGLE);lastMode=REMOTE;currentMode=REMOTE;Serial.println(F("Mode->REMOTE"));} }
        else if (m=="pot")    { lastMode=POT;currentMode=POT;capturedPotAZ=readPotCounts(PIN_POT_AZ);capturedPotEL=readPotCounts(PIN_POT_EL);holdAZ=scaledFeedbackToAngle(jrkAZ.getScaledFeedback(),AZ_MAX_ANGLE);holdEL=scaledFeedbackToAngle(jrkEL.getScaledFeedback(),EL_MAX_ANGLE);azPickedUp=false;elPickedUp=false;Serial.println(F("Mode->POT")); }
        else if (m=="stow")   { lastMode=STOW;currentMode=STOW;Serial.println(F("Mode->STOW")); }
        else if (m=="estop")  { currentMode=ESTOP;Serial.println(F("Mode->ESTOP")); }
        else if (m=="resume") {
            if(sw&&(lastMode==TRACK||lastMode==REMOTE)){currentMode=POT;Serial.println(F("Mode->POT(sw)"));}
            else{currentMode=lastMode;}
        }
    }
    if (currentMode==REMOTE) {
        if (doc.containsKey("az_setpoint")) remoteAZ=constrain(doc["az_setpoint"].as<float>(),-AZ_MAX_ANGLE,AZ_MAX_ANGLE);
        if (doc.containsKey("el_setpoint")) remoteEL=constrain(doc["el_setpoint"].as<float>(),-EL_MAX_ANGLE,EL_MAX_ANGLE);
    }
}


// ════════════════════════════════════════════════════════════════════
//  WIFI / MQTT
// ════════════════════════════════════════════════════════════════════
void startWiFi() {
    Serial.println(F("WiFi: connecting to NETGEAR08..."));
    WiFi.mode(WIFI_STA); WiFi.begin(WIFI_SSID,WIFI_PASSWORD);
}

bool connectMQTT() {
    if (mqttClient.connected()) return true;
    Serial.print(F("MQTT connecting..."));
    if (mqttClient.connect(MQTT_CLIENT_ID,MQTT_USER,MQTT_PASS,TOPIC_STATUS,0,true,"offline")) {
        mqttClient.subscribe(TOPIC_COMMAND);
        mqttClient.publish(TOPIC_STATUS,"online",true);
        Serial.println(F(" connected")); return true;
    }
    Serial.print(F(" failed rc=")); Serial.println(mqttClient.state()); return false;
}


// ════════════════════════════════════════════════════════════════════
//  TELEMETRY
// ════════════════════════════════════════════════════════════════════
void publishTelemetry(float cmdAZ, float cmdEL, float actAZ, float actEL,
                      float errAZ, float errEL, bool jrkOK,
                      uint16_t jrkErrAZ, uint16_t jrkErrEL) {
    StaticJsonDocument<768> doc;
    doc["az_cmd"]          = round(cmdAZ*10.0f)/10.0f;
    doc["el_cmd"]          = round(cmdEL*10.0f)/10.0f;
    doc["az_actual"]       = round(actAZ*10.0f)/10.0f;
    doc["el_actual"]       = round(actEL*10.0f)/10.0f;
    doc["az_error"]        = round(errAZ*10.0f)/10.0f;
    doc["el_error"]        = round(errEL*10.0f)/10.0f;
    doc["mode"]            = (currentMode==POT?"pot":currentMode==REMOTE?"remote":
                              currentMode==STOW?"stow":currentMode==ESTOP?"estop":
                              currentMode==TRACK?"track":currentMode==CAL?"cal":
                              currentMode==STOW_TO_TRACK?"stow_to_track":"unknown");
    doc["jrk_ok"]          = jrkOK;
    doc["hw_manual"]       = (digitalRead(PIN_SWITCH)==LOW);
    doc["uptime_s"]        = millis()/1000;
    doc["wifi_rssi"]       = WiFi.RSSI();
    doc["trk_e_az"]        = round(lastSense.E_az*1000.0f)/1000.0f;
    doc["trk_e_el"]        = round(lastSense.E_el*1000.0f)/1000.0f;
    doc["trk_irrad"]       = round(lastSense.S*1000.0f)/1000.0f;
    doc["trk_valid"]       = lastSense.valid;
    doc["trk_state"]       = (trkState==TRK_ACTIVE?"active":"frozen");
    doc["waiting_for_sun"] = waitingForSun;
    doc["sol_az"]          = round(solAz*10.0f)/10.0f;
    doc["sol_el"]          = round(solEl*10.0f)/10.0f;
    doc["sol_az_dast"]     = round(solAzDast*10.0f)/10.0f;
    doc["sol_el_dast"]     = round(solElDast*10.0f)/10.0f;
    doc["abs_az_error"]    = round((actAZ-solAzDast)*10.0f)/10.0f;
    doc["abs_el_error"]    = round((actEL-solElDast)*10.0f)/10.0f;
    doc["panel_v"]         = round(panelVoltage*100.0f)/100.0f;
    doc["panel_i"]         = round(panelCurrent*1000.0f)/1000.0f;
    doc["panel_w"]         = round(panelPower*100.0f)/100.0f;
    char buffer[768];
    size_t len=serializeJson(doc,buffer);
    mqttClient.publish(TOPIC_TELEMETRY,buffer,len);
}

    Wire.begin(PIN_SDA,PIN_SCL); Wire.setClock(400000);

    lcd.setCursor(0,3); lcd.print(F("                    "));

    Serial.println();
    Serial.println(F("════════════════════════════════════════════════════════"));
    Serial.println(F("  DAST v2.10 — Dual Axis Solar Tracker"));
    Serial.println(F("  Solar position | CAL mode | LDR offsets"));
    Serial.println(F("  Type ? for commands  |  type cal for calibration"));

    // Switch debounce at boot
    delay(50); bool sw1=(digitalRead(PIN_SWITCH)==LOW);
    delay(50); bool sw2=(digitalRead(PIN_SWITCH)==LOW);
    prevSwitchManual=(sw1&&sw2);

    if (prevSwitchManual) {
        capturedPotAZ=readPotCounts(PIN_POT_AZ); capturedPotEL=readPotCounts(PIN_POT_EL);
        holdAZ=scaledFeedbackToAngle(sfAZ,AZ_MAX_ANGLE);
        holdEL=scaledFeedbackToAngle(sfEL,EL_MAX_ANGLE);
        azPickedUp=false; elPickedUp=false;
        currentMode=POT; lastMode=POT;
        Serial.println(F("  Switch: MANUAL"));
    } else {
        currentMode=STOW; lastMode=STOW;
        Serial.println(F("  Switch: AUTO — STOW"));
    }

    Serial.print(F("  LDR offsets: AZ=")); Serial.print(azLdrOffset,4);
    Serial.print(F("  EL=")); Serial.println(elLdrOffset,4);
    Serial.print(F("  Site: ")); Serial.print(siteLat,4);
    Serial.print(F("N  ")); Serial.print(fabsf(siteLon),4);

    { float z=adcPinVolts(PIN_PANEL_CURRENT);
      Serial.print(F("  Current zero pin: ")); Serial.print(z,4); Serial.println(F("V")); }

    mqttClient.setServer(MQTT_SERVER,MQTT_PORT);
    mqttClient.setCallback(onMQTTMessage);
    mqttClient.setBufferSize(768);
    mqttClient.setKeepAlive(60);

    startWiFi();
    lastWifiAttempt=millis();

    lcd.setCursor(0,1);
    lcd.print(prevSwitchManual?F("Mode: MANUAL        "):F("Mode: STOW          "));
    lcd.setCursor(0,3); lcd.print(F("WiFi: connecting... "));

    printHelp();
    Serial.println(F("Running — type cal to calibrate LDR offsets"));
    Serial.println();
}


// ════════════════════════════════════════════════════════════════════
//  MAIN LOOP
// ════════════════════════════════════════════════════════════════════
void loop() {
    static unsigned long lastLoop      = 0;
    static unsigned long lastTelemetry = 0;
    static unsigned long lastLCD       = 0;
    static unsigned long lastCSV       = 0;
    static unsigned long lastSolar     = 0;

    handleSerialTuning();
    if (mqttClient.connected()) mqttClient.loop();

    unsigned long now=millis();
    if (now-lastLoop<LOOP_MS) return;
    lastLoop=now;

    // WiFi watchdog
    bool wifiOK=(WiFi.status()==WL_CONNECTED);
    if (!wifiOK) {
        if (wifiWasConnected) { Serial.println(F("WiFi lost")); wifiWasConnected=false; }
        if (now-lastWifiAttempt>WIFI_RETRY_MS) {
            lastWifiAttempt=now; WiFi.disconnect(); WiFi.begin(WIFI_SSID,WIFI_PASSWORD);
        }
    } else {
        if (!wifiWasConnected) {
            wifiWasConnected=true;
            Serial.print(F("WiFi connected — IP: ")); Serial.println(WiFi.localIP());
            if (!ntpSynced) {
                configTime(GMT_OFFSET,DST_OFFSET,NTP_SERVER);
                struct tm t;
                if (getLocalTime(&t,5000)) {
                    ntpSynced=true;
                    Serial.print(F("NTP synced — ")); Serial.println(getTimeStr());
                    updateSolarPosition();
                } else Serial.println(F("NTP sync failed"));
            }
        }
    }

    // MQTT watchdog
    bool mqttOK=mqttClient.connected();
    if (!mqttOK&&wifiOK&&(now-lastMqttAttempt>MQTT_RECONNECT_MS)) {
        lastMqttAttempt=now; mqttOK=connectMQTT();
    }

    // Update solar position every 30s
    if (ntpSynced && (now-lastSolar>30000UL)) {
        lastSolar=now; updateSolarPosition();
    }

    updatePanelReadings();

    // S updates every loop regardless of mode
    { SenseResult sg=readSensors(); lastSense.S=sg.S; }

    float actAZ=scaledFeedbackToAngle(jrkAZ.getScaledFeedback(),AZ_MAX_ANGLE);
    float actEL=scaledFeedbackToAngle(jrkEL.getScaledFeedback(),EL_MAX_ANGLE);

    bool switchManual=(digitalRead(PIN_SWITCH)==LOW);
    if ( switchManual&&!prevSwitchManual) onSwitchClosed(actAZ,actEL);
    if (!switchManual&& prevSwitchManual) onSwitchOpened();
    prevSwitchManual=switchManual;

    // Pot pickup tracking
    int curPotAZ=0,curPotEL=0;
    if (currentMode==POT||currentMode==CAL) {
        curPotAZ=readPotCounts(PIN_POT_AZ); curPotEL=readPotCounts(PIN_POT_EL);
        if (!azPickedUp&&abs(curPotAZ-capturedPotAZ)>POT_PICKUP_COUNTS) { azPickedUp=true; }
        if (!elPickedUp&&abs(curPotEL-capturedPotEL)>POT_PICKUP_COUNTS) { elPickedUp=true; }
    }

    // Stow-to-track
    if (currentMode==STOW_TO_TRACK) {
        jrkAZ.setTarget(angleToTarget(0.0f,AZ_MAX_ANGLE));
        jrkEL.setTarget(angleToTarget(0.0f,EL_MAX_ANGLE));
        if (!stowConfirmed) {
            if (bothAxesAtStow()) {
                stowConfirmed=true; stowConfirmedMs=now;
                if (waitingForSun) autoStowedAtMs=now;
                Serial.println(F("[STOW_TO_TRACK] Confirmed — 3s"));
            }
        } else if (now-stowConfirmedMs>=STOW_TO_TRACK_DELAY_MS) {
            if (!waitingForSun) { armTracking(); currentMode=TRACK; lastMode=TRACK; Serial.println(F("[STOW_TO_TRACK] -> TRACK")); }
            else { currentMode=TRACK; lastMode=TRACK; Serial.println(F("[STOW_TO_TRACK] -> TRACK (waiting for sun)")); }
        }
        uint16_t eAZ=jrkAZ.getErrorFlagsHalting(), eEL=jrkEL.getErrorFlagsHalting();
        updateNeoPixel((eAZ==0)&&(eEL==0),mqttOK,!wifiOK);
        if (now-lastLCD>LCD_UPDATE_MS) { lastLCD=now; updateLCD(actAZ,actEL,wifiOK,mqttOK); }
        return;
    }

    // Commanded angles
    float cmdAZ=0.0f, cmdEL=0.0f;
    switch(currentMode) {
        case TRACK:
            runTracking(now);
            cmdAZ=trkCmdAZ; cmdEL=trkCmdEL;
            break;
        case CAL:
            if (calTestActive) {
                // Rate controller active with offsets — test mode
                runTracking(now);
                cmdAZ=trkCmdAZ; cmdEL=trkCmdEL;
            } else {
                // Pot control — aim manually
                float pAZ=countsToAngle(curPotAZ,AZ_COUNT_ZERO,AZ_MAX_ANGLE);
                float pEL=countsToAngle(curPotEL,EL_COUNT_ZERO,EL_MAX_ANGLE);
                cmdAZ=azPickedUp?pAZ:holdAZ;
                cmdEL=elPickedUp?pEL:holdEL;
            }
            break;
        case STOW:
            if (waitingForSun) runTracking(now);
            cmdAZ=0.0f; cmdEL=0.0f;
            break;
        case POT: {
            float pAZ=countsToAngle(curPotAZ,AZ_COUNT_ZERO,AZ_MAX_ANGLE);
            float pEL=countsToAngle(curPotEL,EL_COUNT_ZERO,EL_MAX_ANGLE);
            cmdAZ=azPickedUp?pAZ:holdAZ;
            cmdEL=elPickedUp?pEL:holdEL;
            break;
        }
        case REMOTE: cmdAZ=remoteAZ; cmdEL=remoteEL; break;
        case ESTOP:  cmdAZ=heldAZ;   cmdEL=heldEL;   break;
        default:     cmdAZ=0.0f;     cmdEL=0.0f;     break;
    }

    jrkAZ.setTarget(angleToTarget(cmdAZ,AZ_MAX_ANGLE));
    jrkEL.setTarget(angleToTarget(cmdEL,EL_MAX_ANGLE));
    if (currentMode!=ESTOP) { heldAZ=actAZ; heldEL=actEL; }

    float errAZ=actAZ-cmdAZ, errEL=actEL-cmdEL;

    uint16_t jrkErrAZ=jrkAZ.getErrorFlagsHalting();
    uint16_t jrkErrEL=jrkEL.getErrorFlagsHalting();
    if (jrkErrAZ) { Serial.print(F("AZ Jrk err: 0x")); Serial.println(jrkErrAZ,HEX); }
    if (jrkErrEL) { Serial.print(F("EL Jrk err: 0x")); Serial.println(jrkErrEL,HEX); }
    bool jrkOK=(jrkErrAZ==0)&&(jrkErrEL==0);

    dashboardActive=(lastHeartbeatMs>0)&&((now-lastHeartbeatMs)<HEARTBEAT_TIMEOUT_MS);
    updateNeoPixel(jrkOK,mqttOK,!wifiOK);

    if (now-lastLCD>LCD_UPDATE_MS) { lastLCD=now; updateLCD(actAZ,actEL,wifiOK,mqttOK); }

    if (mqttOK&&(now-lastTelemetry>TELEMETRY_MS)) {
        lastTelemetry=now;
        publishTelemetry(cmdAZ,cmdEL,actAZ,actEL,errAZ,errEL,jrkOK,jrkErrAZ,jrkErrEL);
    }

    // CSV logging
    if (csvLogging&&(now-lastCSV>TELEMETRY_MS)) {
        lastCSV=now;
        unsigned long darkSec=(!lastSense.valid&&darkStartMs>0)?(now-darkStartMs)/1000:0;
        String modeStr;
        if      (currentMode==POT)           modeStr="pot";
        else if (currentMode==REMOTE)        modeStr="remote";
        else if (currentMode==STOW)          modeStr="stow";
        else if (currentMode==ESTOP)         modeStr="estop";
        else if (currentMode==TRACK)         modeStr="track";
        else if (currentMode==CAL&&calTestActive) modeStr="cal_test";
        else if (currentMode==CAL)           modeStr="cal";
        else if (currentMode==STOW_TO_TRACK) modeStr="stowin";
        else                                 modeStr="unk";

        Serial.print(getTimestampCSV());     Serial.print(",");
        Serial.print(modeStr);               Serial.print(",");
        Serial.print(cmdAZ,2);               Serial.print(",");
        Serial.print(actAZ,2);               Serial.print(",");
        Serial.print(errAZ,2);               Serial.print(",");
        Serial.print(cmdEL,2);               Serial.print(",");
        Serial.print(actEL,2);               Serial.print(",");
        Serial.print(errEL,2);               Serial.print(",");
        Serial.print(lastSense.E_az,3);      Serial.print(",");
        Serial.print(lastSense.E_el,3);      Serial.print(",");
        Serial.print(lastSense.S,3);         Serial.print(",");
        Serial.print(lastSense.valid?1:0);   Serial.print(",");
        Serial.print(trkState==TRK_ACTIVE?"active":"frozen"); Serial.print(",");
        Serial.print(panelVoltage,2);        Serial.print(",");
        Serial.print(panelCurrent,3);        Serial.print(",");
        Serial.print(panelPower,2);          Serial.print(",");
        Serial.print(solAz,2);               Serial.print(",");
        Serial.print(solEl,2);               Serial.print(",");
        Serial.print(solAzDast,2);           Serial.print(",");
        Serial.print(solElDast,2);           Serial.print(",");
        Serial.print(actAZ-solAzDast,2);     Serial.print(",");
        Serial.print(actEL-solElDast,2);     Serial.print(",");
        Serial.print(now/1000);              Serial.print(",");
        Serial.print(darkSec);               Serial.print(",");
        Serial.println(waitingForSun?1:0);
    }
