// ════════════════════════════════════════════════════════════════════
//  DAST — Dual Axis Solar Tracker
//  Nano ESP32 + Jrk G2 I2C + Mosquitto MQTT (port 1883, password auth)
//
//  Hardware:
//    Arduino Nano ESP32 (ESP32-S3)
//    Jrk G2 21v3 Azimuth  — I2C address 11 — travel ±90°
//    Jrk G2 21v3 Elevation — I2C address 12 — travel ±84°
//    2.2kΩ I2C pull-up resistors to 3.3V (no level shifter)
//    AZ control pot → A6   EL control pot → A7
//    Hardware lock switch → D4 (closed = forced POT mode)
//    Green LED → D2   Red LED → D3   Lock LED → D5
//
//  Operating modes:
//    POT    — pots command panel position
//    REMOTE — Node-RED setpoints command panel via MQTT
//    STOW   — both axes drive to 0° (default on startup)
//    ESTOP  — holds last position, blocks all commands except RESUME
//
//  MQTT topics:
//    dast/telemetry   ESP32 → Node-RED  (JSON: position, error, status)
//    dast/command     Node-RED → ESP32  (JSON: mode, setpoints, LEDs)
//    dast/status      ESP32 → Node-RED  (online/offline)
// ════════════════════════════════════════════════════════════════════

#include <Wire.h>
#include <JrkG2.h>
#include <WiFi.h>
#include <PubSubClient.h>
#include <ArduinoJson.h>   // version 6.x

// ── WiFi credentials ──────────────────────────────────────────────
const char* WIFI_SSID     = "NETGEAR08";
const char* WIFI_PASSWORD = "pinktrumpet867";

// ── MQTT broker (Mosquitto on AWS EC2) ────────────────────────────
const char* MQTT_SERVER    = "3.130.168.101";
const int   MQTT_PORT      = 1883;
const char* MQTT_USER      = "dast";
const char* MQTT_PASS      = "unREALGems@419";
const char* MQTT_CLIENT_ID = "DAST-NanoESP32";

// ── MQTT topics ───────────────────────────────────────────────────
const char* TOPIC_TELEMETRY = "dast/telemetry";
const char* TOPIC_COMMAND   = "dast/command";
const char* TOPIC_STATUS    = "dast/status";

// ── Jrk G2 controllers ───────────────────────────────────────────
JrkG2I2C jrkAZ(11);
JrkG2I2C jrkEL(12);

// ── Pin definitions ───────────────────────────────────────────────
const int PIN_POT_AZ    = A6;
const int PIN_POT_EL    = A7;
const int PIN_LED_GREEN = D2;
const int PIN_LED_RED   = D3;
const int PIN_MODE_BTN  = D4;   // Hardware lock switch — GND = POT locked
const int PIN_LED_LOCK  = D5;   // Amber LED — blinks when hardware lock active

// ── ADC configuration ─────────────────────────────────────────────
const int NUM_SAMPLES = 32;

// ── Pot calibration ───────────────────────────────────────────────
const int POT_COUNT_MIN  = 0;
const int POT_COUNT_MAX  = 4095;
const int AZ_COUNT_ZERO  = 1908;   // AZ pot count at DAST horizontal
const int EL_COUNT_ZERO  = 1923;   // EL pot count at DAST horizontal

// ── Axis travel limits ────────────────────────────────────────────
const float AZ_MAX_ANGLE = 90.0f;
const float EL_MAX_ANGLE = 84.0f;

// ── Jrk range ─────────────────────────────────────────────────────
const int JRK_MIN = 0;
const int JRK_MAX = 4095;

// ── Loop timing ───────────────────────────────────────────────────
const unsigned long LOOP_MS           = 100;
const unsigned long MQTT_RECONNECT_MS = 5000;
const unsigned long TELEMETRY_MS      = 500;
const unsigned long SERIAL_MS         = 3000;

// ── Operating mode ────────────────────────────────────────────────
enum Mode { POT, REMOTE, STOW, ESTOP };
Mode currentMode = STOW;
Mode lastMode    = STOW;

// ── Remote setpoints ─────────────────────────────────────────────
float remoteAZ = 0.0f;
float remoteEL = 0.0f;

// ── ESTOP held position ───────────────────────────────────────────
float heldAZ = 0.0f;
float heldEL = 0.0f;

// ── MQTT client ───────────────────────────────────────────────────
WiFiClient   wifiClient;
PubSubClient mqttClient(wifiClient);

// ════════════════════════════════════════════════════════════════════
//  CONVERSION FUNCTIONS
// ════════════════════════════════════════════════════════════════════

int readPotCounts(int pin) {
    long sum = 0;
    for (int i = 0; i < NUM_SAMPLES; i++) {
        sum += analogRead(pin);
        delayMicroseconds(200);
    }
    return (int)(sum / NUM_SAMPLES);
}

float countsToAngle(int counts, int zeroCount, float maxAngle) {
    float angle;
    if (counts <= zeroCount) {
        int span = zeroCount - POT_COUNT_MIN;
        if (span == 0) return 0.0f;
        angle = ((float)(counts - zeroCount) / (float)span) * maxAngle;
    } else {
        int span = POT_COUNT_MAX - zeroCount;
        if (span == 0) return 0.0f;
        angle = ((float)(counts - zeroCount) / (float)span) * maxAngle;
    }
    return constrain(angle, -maxAngle, maxAngle);
}

uint16_t angleToTarget(float angle, float maxAngle) {
    float clamped = constrain(angle, -maxAngle, maxAngle);
    float t = (clamped + maxAngle) / (2.0f * maxAngle);
    return (uint16_t)constrain((int)(t * (float)JRK_MAX), JRK_MIN, JRK_MAX);
}

float scaledFeedbackToAngle(uint16_t sf, float maxAngle) {
    return ((float)sf / (float)JRK_MAX) * (2.0f * maxAngle) - maxAngle;
}

// ════════════════════════════════════════════════════════════════════
//  MQTT COMMAND CALLBACK
// ════════════════════════════════════════════════════════════════════

void onMQTTMessage(char* topic, byte* payload, unsigned int length) {
    if (strcmp(topic, TOPIC_COMMAND) != 0) return;

    StaticJsonDocument<256> doc;
    DeserializationError err = deserializeJson(doc, payload, length);
    if (err) {
        Serial.print("JSON parse error: ");
        Serial.println(err.c_str());
        return;
    }

    // Mode change — blocked if hardware lock switch is closed
    if (doc.containsKey("mode")) {
        if (digitalRead(PIN_MODE_BTN) == HIGH) {
            String m = doc["mode"].as<String>();
            if (m == "remote") {
                uint16_t sfAZ = jrkAZ.getScaledFeedback();
                uint16_t sfEL = jrkEL.getScaledFeedback();
                remoteAZ = scaledFeedbackToAngle(sfAZ, AZ_MAX_ANGLE);
                remoteEL = scaledFeedbackToAngle(sfEL, EL_MAX_ANGLE);
                lastMode    = REMOTE;
                currentMode = REMOTE;
                Serial.println("Mode → REMOTE (holding current position)");
                Serial.print("  AZ hold: "); Serial.print(remoteAZ, 1);
                Serial.print("°  EL hold: "); Serial.print(remoteEL, 1);
                Serial.println("°");
            }
            else if (m == "pot") {
                lastMode    = POT;
                currentMode = POT;
                Serial.println("Mode → POT");
            }
            else if (m == "stow") {
                lastMode    = STOW;
                currentMode = STOW;
                Serial.println("Mode → STOW");
            }
            else if (m == "estop") {
                currentMode = ESTOP;
                Serial.println("Mode → ESTOP");
            }
            else if (m == "resume") {
                currentMode = lastMode;
                Serial.print("Mode → RESUME (returning to ");
                Serial.print(lastMode == POT ? "POT" : lastMode == REMOTE ? "REMOTE" : "STOW");
                Serial.println(")");
            }
        } else {
            Serial.println("Hardware lock active — remote mode command ignored");
        }
    }

    // Setpoints
    if (doc.containsKey("az_setpoint"))
        remoteAZ = constrain(doc["az_setpoint"].as<float>(), -AZ_MAX_ANGLE, AZ_MAX_ANGLE);
    if (doc.containsKey("el_setpoint"))
        remoteEL = constrain(doc["el_setpoint"].as<float>(), -EL_MAX_ANGLE, EL_MAX_ANGLE);
}

// ════════════════════════════════════════════════════════════════════
//  WiFi CONNECTION
// ════════════════════════════════════════════════════════════════════

void connectWiFi() {
    Serial.print("Connecting to WiFi");
    WiFi.mode(WIFI_STA);
    WiFi.begin(WIFI_SSID, WIFI_PASSWORD);
    int attempts = 0;
    while (WiFi.status() != WL_CONNECTED && attempts < 40) {
        delay(500);
        Serial.print(".");
        attempts++;
    }
    if (WiFi.status() == WL_CONNECTED) {
        Serial.println(" connected");
        Serial.print("  IP: "); Serial.println(WiFi.localIP());
    } else {
        Serial.println(" FAILED — will retry in loop");
    }
}

// ════════════════════════════════════════════════════════════════════
//  MQTT CONNECTION
// ════════════════════════════════════════════════════════════════════

bool connectMQTT() {
    if (mqttClient.connected()) return true;
    Serial.print("Connecting to MQTT broker...");
    if (mqttClient.connect(MQTT_CLIENT_ID, MQTT_USER, MQTT_PASS,
                           TOPIC_STATUS, 0, true, "offline")) {
        mqttClient.subscribe(TOPIC_COMMAND);
        mqttClient.publish(TOPIC_STATUS, "online", true);
        Serial.println(" connected");
        return true;
    }
    Serial.print(" failed rc=");
    Serial.println(mqttClient.state());
    return false;
}

// ════════════════════════════════════════════════════════════════════
//  PUBLISH TELEMETRY
// ════════════════════════════════════════════════════════════════════

void publishTelemetry(float cmdAZ, float cmdEL,
                      float actAZ, float actEL,
                      float errAZ, float errEL,
                      bool  jrkOK,
                      uint16_t jrkErrAZ, uint16_t jrkErrEL) {
    StaticJsonDocument<384> doc;
    doc["az_cmd"]    = round(cmdAZ * 10.0f) / 10.0f;
    doc["el_cmd"]    = round(cmdEL * 10.0f) / 10.0f;
    doc["az_actual"] = round(actAZ * 10.0f) / 10.0f;
    doc["el_actual"] = round(actEL * 10.0f) / 10.0f;
    doc["az_error"]  = round(errAZ * 10.0f) / 10.0f;
    doc["el_error"]  = round(errEL * 10.0f) / 10.0f;
    doc["mode"]      = (currentMode == POT    ? "pot"    :
                        currentMode == REMOTE ? "remote" :
                        currentMode == STOW   ? "stow"   :
                        currentMode == ESTOP  ? "estop"  : "unknown");
    doc["jrk_ok"]    = jrkOK;
    doc["jrk_az_ok"] = (jrkErrAZ == 0);
    doc["jrk_el_ok"] = (jrkErrEL == 0);
    doc["led_green"] = (digitalRead(PIN_LED_GREEN) == HIGH);
    doc["led_red"]   = (digitalRead(PIN_LED_RED)   == HIGH);
    doc["hw_lock"]   = (digitalRead(PIN_MODE_BTN)  == LOW);
    doc["uptime_s"]  = millis() / 1000;
    doc["wifi_rssi"] = WiFi.RSSI();

    char buffer[384];
    size_t len = serializeJson(doc, buffer);
    mqttClient.publish(TOPIC_TELEMETRY, buffer, len);
}

// ════════════════════════════════════════════════════════════════════
//  SETUP
// ════════════════════════════════════════════════════════════════════

void setup() {
    Serial.begin(115200);
    delay(1000);

    // ── Pin setup ─────────────────────────────────────────────────
    pinMode(PIN_LED_GREEN, OUTPUT);
    pinMode(PIN_LED_RED,   OUTPUT);
    pinMode(PIN_LED_LOCK,  OUTPUT);
    pinMode(PIN_MODE_BTN,  INPUT_PULLUP);
    digitalWrite(PIN_LED_GREEN, LOW);
    digitalWrite(PIN_LED_RED,   HIGH);
    digitalWrite(PIN_LED_LOCK,  LOW);

    // ── ADC setup ─────────────────────────────────────────────────
    analogReadResolution(12);
    analogSetAttenuation(ADC_11db);

    // ── I2C setup ─────────────────────────────────────────────────
    Wire.begin();
    Wire.setClock(400000);

    // ── Clear Jrk startup errors ──────────────────────────────────
    jrkAZ.getErrorFlagsHalting();
    jrkEL.getErrorFlagsHalting();

    Serial.println();
    Serial.println("════════════════════════════════════════════════════");
    Serial.println("  DAST — Dual Axis Solar Tracker");
    Serial.println("  Mosquitto MQTT  |  port 1883  |  password auth");
    Serial.println("════════════════════════════════════════════════════");

    Serial.println("Checking Jrk controllers...");
    uint16_t sfAZ = jrkAZ.getScaledFeedback();
    uint16_t sfEL = jrkEL.getScaledFeedback();
    Serial.print("  AZ (addr 11): "); Serial.println(sfAZ <= JRK_MAX ? "OK" : "NO RESPONSE");
    Serial.print("  EL (addr 12): "); Serial.println(sfEL <= JRK_MAX ? "OK" : "NO RESPONSE");

    connectWiFi();

    mqttClient.setServer(MQTT_SERVER, MQTT_PORT);
    mqttClient.setCallback(onMQTTMessage);
    mqttClient.setBufferSize(512);
    mqttClient.setKeepAlive(60);
    connectMQTT();

    bool systemOK = mqttClient.connected();
    digitalWrite(PIN_LED_GREEN, systemOK ? HIGH : LOW);
    digitalWrite(PIN_LED_RED,   systemOK ? LOW  : HIGH);

    Serial.println();
    Serial.println("Running — default mode: STOW");
    Serial.println("Use dashboard or hardware switch to change mode");
    Serial.println();
}

// ════════════════════════════════════════════════════════════════════
//  MAIN LOOP
// ════════════════════════════════════════════════════════════════════

void loop() {
    static unsigned long lastLoop      = 0;
    static unsigned long lastReconnect = 0;
    static unsigned long lastTelemetry = 0;

    // ── Process incoming MQTT messages ────────────────────────────
    if (mqttClient.connected()) mqttClient.loop();

    // ── Enforce 100ms loop rate ───────────────────────────────────
    unsigned long now = millis();
    if (now - lastLoop < LOOP_MS) return;
    lastLoop = now;

    // ── Maintain WiFi ─────────────────────────────────────────────
    if (WiFi.status() != WL_CONNECTED) {
        Serial.println("WiFi lost — reconnecting");
        connectWiFi();
    }

    // ── Maintain MQTT (non-blocking) ──────────────────────────────
    bool mqttOK = mqttClient.connected();
    if (!mqttOK && (now - lastReconnect > MQTT_RECONNECT_MS)) {
        lastReconnect = now;
        mqttOK = connectMQTT();
    }

    // ── Hardware POT lock switch ──────────────────────────────────
    bool potLockActive = (digitalRead(PIN_MODE_BTN) == LOW);
    if (potLockActive) {
        digitalWrite(PIN_LED_LOCK, (millis() / 500) % 2);
        if (currentMode != POT) {
            currentMode = POT;
            lastMode    = POT;
            Serial.println("Hardware lock active — Mode forced to POT");
        }
    } else {
        digitalWrite(PIN_LED_LOCK, LOW);
    }

    // ── Determine commanded angles ────────────────────────────────
    float cmdAZ, cmdEL;
    switch (currentMode) {
        case ESTOP:
            cmdAZ = heldAZ;
            cmdEL = heldEL;
            break;
        case POT:
            cmdAZ = countsToAngle(readPotCounts(PIN_POT_AZ), AZ_COUNT_ZERO, AZ_MAX_ANGLE);
            cmdEL = countsToAngle(readPotCounts(PIN_POT_EL), EL_COUNT_ZERO, EL_MAX_ANGLE);
            break;
        case REMOTE:
            cmdAZ = remoteAZ;
            cmdEL = remoteEL;
            break;
        case STOW:
        default:
            cmdAZ = 0.0f;
            cmdEL = 0.0f;
            break;
    }

    // ── Send targets to Jrk controllers ──────────────────────────
    jrkAZ.setTarget(angleToTarget(cmdAZ, AZ_MAX_ANGLE));
    jrkEL.setTarget(angleToTarget(cmdEL, EL_MAX_ANGLE));

    // ── Read actual positions ─────────────────────────────────────
    float actAZ = scaledFeedbackToAngle(jrkAZ.getScaledFeedback(), AZ_MAX_ANGLE);
    float actEL = scaledFeedbackToAngle(jrkEL.getScaledFeedback(), EL_MAX_ANGLE);

    if (currentMode != ESTOP) {
        heldAZ = actAZ;
        heldEL = actEL;
    }

    float errAZ = actAZ - cmdAZ;
    float errEL = actEL - cmdEL;

    // ── Read and clear Jrk error flags ────────────────────────────
    uint16_t jrkErrAZ = jrkAZ.getErrorFlagsHalting();
    uint16_t jrkErrEL = jrkEL.getErrorFlagsHalting();
    if (jrkErrAZ) { Serial.print("AZ Jrk error: 0x"); Serial.println(jrkErrAZ, HEX); }
    if (jrkErrEL) { Serial.print("EL Jrk error: 0x"); Serial.println(jrkErrEL, HEX); }
    bool jrkOK = (jrkErrAZ == 0) && (jrkErrEL == 0);

    // ── Update LEDs ───────────────────────────────────────────────
    bool systemOK = jrkOK && mqttOK;
    digitalWrite(PIN_LED_GREEN, systemOK ? HIGH : LOW);
    digitalWrite(PIN_LED_RED,   systemOK ? LOW  : HIGH);

    // ── Publish telemetry ─────────────────────────────────────────
    if (mqttOK && (now - lastTelemetry > TELEMETRY_MS)) {
        lastTelemetry = now;
        publishTelemetry(cmdAZ, cmdEL, actAZ, actEL,
                         errAZ, errEL, jrkOK,
                         jrkErrAZ, jrkErrEL);
    }

    // ── Serial status output ──────────────────────────────────────
    static unsigned long lastSerial = 0;
    static unsigned int  lineCount  = 0;

    if (now - lastSerial > SERIAL_MS) {
        lastSerial = now;

        if (lineCount % 10 == 0) {
            Serial.println();
            Serial.println("━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━");
            Serial.println("  DAST Status");
            Serial.println("━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━");
        }
        lineCount++;

        Serial.print("  Mode      : ");
        Serial.println(currentMode == POT    ? "POT"    :
                       currentMode == REMOTE ? "REMOTE" :
                       currentMode == STOW   ? "STOW"   :
                       currentMode == ESTOP  ? "ESTOP"  : "UNKNOWN");

        Serial.print("  HW Lock   : ");
        Serial.println(potLockActive ? "ACTIVE" : "off");

        Serial.print("  MQTT      : ");
        Serial.println(mqttOK ? "Connected" : "Disconnected");

        Serial.print("  Azimuth   : cmd ");
        Serial.print(cmdAZ, 1);
        Serial.print("°  actual ");
        Serial.print(actAZ, 1);
        Serial.print("°  error ");
        Serial.print(errAZ, 1);
        Serial.println("°");

        Serial.print("  Elevation : cmd ");
        Serial.print(cmdEL, 1);
        Serial.print("°  actual ");
        Serial.print(actEL, 1);
        Serial.print("°  error ");
        Serial.print(errEL, 1);
        Serial.println("°");

        Serial.print("  Jrk       : ");
        Serial.println(jrkOK ? "OK" : "ERROR");

        Serial.print("  LED       : ");
        Serial.println(systemOK ? "Green (system OK)" : "Red (fault)");

        Serial.print("  Uptime    : ");
        Serial.print(millis() / 1000);
        Serial.println(" s");

        Serial.print("  WiFi RSSI : ");
        Serial.print(WiFi.RSSI());
        Serial.println(" dBm");

        Serial.println("─────────────────────────────────────────────────────────────");
    }
}
